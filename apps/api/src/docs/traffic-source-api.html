Authentication

All API requests require authentication via the X-API-Key header

Base URL https://api.motivbuy.com/api/v1/source
Request Header 
X-API-Key: your_api_key_here
Content-Type: application/json
Security Warning Keep your API key secure. Never expose it in client-side code or public repositories.

Get Available Tasks

Fetch tasks for a user based on targeting criteria

POST /api/v1/source/tasks
Returns a list of available tasks filtered by user targeting parameters. Tasks are matched against order requirements.
Request Body
Parameter Type Required Description
userId integer Required User's Telegram ID
limit integer Optional Max tasks to return (1-20, default: 5)
Targeting Parameters (Optional)

Pass user data to filter tasks by targeting requirements. Only tasks matching ALL provided criteria will be returned.

Parameter Type Example Description
gender string male User gender: male or female
age integer 25 User age (13-100)
country string US ISO 3166-1 alpha-2 country code
region string CA Region or state code
city string Los Angeles City name
language string en ISO 639-1 language code
Response - Task Object
Field Type Description
taskId string Unique task identifier (use for /tasks/check)
orderId string Order identifier
action string subscribe | join | view | react | comment | start
price number Reward amount in USDT
link string Target link (Telegram channel/bot)
name string Target name
remainingSlots integer Available slots remaining
Example

curl -X POST https://api.motivbuy.com/api/v1/source/tasks \
  -H "X-API-Key: your_api_key_here" \
  -H "Content-Type: application/json" \

    "userId": 123456789,

Go 
package main

import (
    "bytes"
    "encoding/json"
    "net/http"
    "os"
)

func getTasks(userID int64) (map[string]interface{}, error) {
    payload := map[string]interface{}{
        "userId":   userID,
        "limit":    10,
        "gender":   "male",
        "age":      25,
        "country":  "US",
        "language": "en",
    }

    body, _ := json.Marshal(payload)
    req, _ := http.NewRequest("POST",
        "https://api.motivbuy.com/api/v1/source/tasks",
        bytes.NewBuffer(body))

    req.Header.Set("X-API-Key", os.Getenv("API_KEY"))
    req.Header.Set("Content-Type", "application/json")

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    return result, nil
}

// Usage
func main() {
    tasks, _ := getTasks(123456789)
    fmt.Println(tasks)
}

Check Task Status

Verify if a user has completed a specific task

POST /api/v1/source/tasks/check
Check whether a task has been completed by the user. Use the taskId returned from the /tasks endpoint.
Request Body
Parameter Type Required Description
taskId string Required Task ID from /tasks response
Response
Field Type Description
status string not_started | pending | completed | failed
canSubmit boolean Whether the task can still be submitted
completedAt string ISO 8601 timestamp when completed
Example
cURL 
curl -X POST https://api.motivbuy.com/api/v1/source/tasks/check \
  -H "X-API-Key: your_api_key_here" \
  -H "Content-Type: application/json" \
  -d '{"taskId": "ORD-01HXYZ-USER-123456789"}'
TypeScript 
interface TaskStatusResponse {
  status: 'not_started' | 'pending' | 'completed' | 'failed';
  canSubmit: boolean;
  completedAt?: string;
}

const checkTaskStatus = async (taskId: string): Promise<TaskStatusResponse> => {

      'X-API-Key': process.env.API_KEY,
      'Content-Type': 'application/json',

    body: JSON.stringify({ taskId }),
  });

  return response.json();
};

// Usage
const status = await checkTaskStatus('ORD-01HXYZ-USER-123456789');
if (status.status === 'completed') {
  console.log('Task completed at:', status.completedAt);
}

package main

import (
    "bytes"
    "encoding/json"
    "net/http"
    "os"
)

type TaskStatusResponse struct {
    Status      string `json:"status"`
    CanSubmit   bool   `json:"canSubmit"`
    CompletedAt string `json:"completedAt,omitempty"`
}

func checkTaskStatus(taskID string) (*TaskStatusResponse, error) {
    payload := map[string]string{"taskId": taskID}
    body, _ := json.Marshal(payload)

    req, _ := http.NewRequest("POST",
        "https://api.motivbuy.com/api/v1/source/tasks/check",
        bytes.NewBuffer(body))

    req.Header.Set("X-API-Key", os.Getenv("API_KEY"))
    req.Header.Set("Content-Type", "application/json")

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var result TaskStatusResponse
    json.NewDecoder(resp.Body).Decode(&result)
    return &result, nil
}

// Usage
func main() {
    status, _ := checkTaskStatus("ORD-01HXYZ-USER-123456789")
    if status.Status == "completed" {
        fmt.Println("Task completed at:", status.CompletedAt)
    }
}

Error Handling

All errors return a consistent JSON format

Error Response Format 
{
  "statusCode": 401,
  "message": "Invalid API key",
  "error": "Unauthorized"
}
200

Success

Request completed successfully

400

Bad Request

Invalid parameters or validation failed

401

Unauthorized

Invalid or missing API key

403

Forbidden

API key is inactive or suspended

429

Too Many Requests

Rate limit exceeded - wait and retry

500

Server Error

Internal error - please retry later

Rate Limits

API rate limiting is applied per API key

10,000 requests per minute That's ~166 requests per second sustained throughput per API key.

When rate limited:

  • API returns HTTP 429 Too Many Requests
  • Wait for the 1-minute window to reset
  • Implement exponential backoff for retries
